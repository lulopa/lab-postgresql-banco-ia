6. Ejecutar una transferencia válida (Débito y crédito atómicos) 
-- 6.1 Ejecución de la transferencia entre la Cuenta 1 y la Cuenta 2 
@delimiter /; 
DO $$ 
DECLARE 
    v_cli1 BIGINT; 
    v_cli2 BIGINT; 
    v_cta_orig BIGINT; 
    v_cta_dest BIGINT; 
    v_num1 VARCHAR(20); 
    v_num2 VARCHAR(20); 
    v_tx_deposito BIGINT; 
    v_tx_transferencia BIGINT; 
BEGIN 
    -- 1. Crear 2 clientes de prueba 
    CALL banco_core.sp_crear_cliente('NATURAL', 1, 1, '90000001', 'ALBERTO', 'LOPEZ', '1985-01-01', NULL, NULL, v_cli1); 
    CALL banco_core.sp_crear_cliente('NATURAL', 1, 1, '90000002', 'BEATRIZ', 'GOMEZ', '1990-02-02', NULL, NULL, v_cli2); 
 
    -- 2. Crear y activar 2 cuentas 
    CALL banco_core.sp_crear_cuenta(v_cli1, 1, 1, 1, v_cta_orig, v_num1); 
    CALL banco_core.sp_crear_cuenta(v_cli2, 1, 1, 1, v_cta_dest, v_num2); 
    CALL banco_core.sp_activar_cuenta(v_cta_orig, 1, 'Activación para prueba de transferencia'); 
    CALL banco_core.sp_activar_cuenta(v_cta_dest, 1, 'Activación para prueba de transferencia'); 
 
    -- 3. Fonder la cuenta de origen con $200.000 COP 
    CALL banco_core.sp_consignar(uuid_generate_v4(), v_cta_orig, 200000.00, 1, v_tx_deposito); 
 
    -- 4. Ejecutar la transferencia de $50.000 COP 
    CALL banco_core.sp_transferir(uuid_generate_v4(), v_cta_orig, v_cta_dest, 50000.00, 1, v_tx_transferencia); 
 
    RAISE NOTICE 'Transferencia ejecutada con éxito. ID Transacción: %', v_tx_transferencia; 
END $$; 
/ 
@delimiter ;/ 
 
-- 6.2 Consulta de verificación de la Partida Doble en el Ledger 
SELECT 
    t.id_transaccion, 
    tt.nombre AS operacion, 
    t.monto AS monto_transaccion, 
    ac.num_linea, 
    ac.codigo_cuenta_puc, 
    ac.tipo_movimiento, 
    ac.monto AS monto_asiento 
FROM banco_core.transaccion t 
JOIN banco_core.tipo_transaccion tt ON t.id_tipo_transaccion = tt.id_tipo_transaccion 
JOIN banco_core.asiento_contable ac ON t.id_transaccion = ac.id_transaccion 
WHERE t.id_transaccion = (SELECT MAX(id_transaccion) FROM banco_core.transaccion) 
ORDER BY ac.num_linea ASC;

7. Revertir una transacción mediante operación compensatoria (Preservar histórico) 
-- 7.1 Reversión de la última transferencia ejecutada 
@delimiter /; 
DO $$ 
DECLARE 
    v_tx_original BIGINT; 
    v_id_reverso BIGINT; 
BEGIN 
    -- Obtener la última transferencia exitosa 
    SELECT MAX(id_transaccion) INTO v_tx_original 
    FROM banco_core.transaccion 
    WHERE id_tipo_transaccion = 3 AND id_estado_transaccion = 2; 
 
    IF v_tx_original IS NULL THEN 
        RAISE EXCEPTION 'No se encontró una transferencia en estado POSTED para reversar'; 
    END IF; 
 
    -- Ejecutar el reverso compensatorio 
    CALL banco_core.sp_reversar_transaccion( 
        v_tx_original, 
        1, 
        'Reverso compensatorio de prueba de auditoría G3', 
        v_id_reverso 
    ); 
 
    RAISE NOTICE 'Transacción % reversada exitosamente mediante reverso %', v_tx_original, v_id_reverso; 
END $$; 
/ 
@delimiter ;/ 

 
-- 7.2 Consulta de verificación de inmutabilidad e historial preservado 
SELECT 
    t.id_transaccion, 
    tt.nombre AS tipo_transaccion, 
    et.codigo AS estado_transaccion, 
    t.monto, 
    t.fecha_transaccion 
FROM banco_core.transaccion t 
JOIN banco_core.tipo_transaccion tt ON t.id_tipo_transaccion = tt.id_tipo_transaccion 
JOIN banco_core.estado_transaccion et ON t.id_estado_transaccion = et.id_estado_transaccion 
ORDER BY t.id_transaccion DESC
